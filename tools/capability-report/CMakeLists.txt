# Portable argument parsing, rendering, and file output; tests link it with injected reports.
add_library(catro_report_cli STATIC report_cli.cpp)
target_include_directories(catro_report_cli PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(catro_report_cli PUBLIC Catro::Reporting)
catro_enable_warnings(catro_report_cli)

if(WIN32 AND MSVC)
    add_executable(catro_capability_report main.cpp)
    set_target_properties(catro_capability_report PROPERTIES OUTPUT_NAME "catro-capability-report")
    target_link_libraries(catro_capability_report PRIVATE catro_report_cli Catro::Windows)
    target_compile_definitions(catro_capability_report PRIVATE NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00)
    add_dependencies(catro_capability_report catro_capability_probe)
    catro_enable_warnings(catro_capability_report)
elseif(APPLE)
    add_executable(catro_capability_report main_macos.mm)
    set_target_properties(catro_capability_report PROPERTIES OUTPUT_NAME "catro-capability-report")
    target_link_libraries(catro_capability_report PRIVATE catro_report_cli Catro::Macos "-framework Foundation")
    target_compile_options(catro_capability_report PRIVATE -fobjc-arc)
    add_dependencies(catro_capability_report catro_capability_probe)
    catro_enable_warnings(catro_capability_report)
endif()
